# Official-repository packages for MyArch.
#
# Single source of truth: install.sh reads this file, and the docs point here
# rather than listing names in prose. One package per line; blank lines and
# lines starting with # are ignored. Trailing comments are stripped.

# ── Compositor & session ────────────────────────────────────────────────────
hyprland                # the compositor itself (Lua config needs >= 0.55)
hyprlock                # lock screen (hypr/hyprlock.conf)
hypridle                # idle daemon (hypr/hypridle.conf, Coffee Mode toggle)
waybar                  # status bar, must be built with Hyprland support
swaync                  # notifications; started from autostart.lua
swayosd                 # volume/brightness OSD, themed by wallust

# ── Terminal, launcher, file manager ────────────────────────────────────────
kitty                   # terminal (lib.lua M.terminal)
rofi-wayland            # launcher, clipboard, power menu, window switcher
thunar                  # file manager (lib.lua M.fileManager)
xarchiver               # archive manager: Thunar's Extract Here / Create Archive
thunar-archive-plugin   # puts those entries in Thunar's right-click menu
zip                     # xarchiver shells out to it to CREATE .zip (unzip only reads)
tumbler                 # thumbnail service; without it Thunar shows generic icons
webp-pixbuf-loader      # so .webp gets thumbnails like png/jpg

# ── Login screen ────────────────────────────────────────────────────────────
# qylock's themes are SDDM themes, so SDDM is the login manager. Note that
# Arch's sddm hard-depends on xorg-server: the greeter runs under X even though
# the session itself is Wayland. greetd + noctalia-greeter stay installed as the
# fallback -- see sddm/README.md.
sddm                    # display manager; sddm/install-theme.sh adds the theme
gst-plugins-base        # video playback for the animated theme background
gst-plugins-good        # same

# ── Screenshot, recording, clipboard ────────────────────────────────────────
grim                    # screenshots (Print, SUPER+P)
slurp                   # region select for screenshots and recording
wl-clipboard            # wl-copy / wl-paste
libnotify               # notify-send, used by the scripts and ocr-snipper
cliphist                # clipboard history (SUPER+V)
wf-recorder             # screen recording (SUPER+SHIFT+P)

# ── Media & hardware keys ───────────────────────────────────────────────────
playerctl               # media keys
brightnessctl           # waybar backlight scroll
power-profiles-daemon   # powerprofilesctl, the SUPER+SHIFT+G menu
intel-gpu-tools         # intel_gpu_top, for waybar's custom/gpu module

# ── Scripts & waybar modules ────────────────────────────────────────────────
fd                      # SUPER+SHIFT+E file finder (hard requirement)
jq                      # SUPER+Tab window switcher
pacman-contrib          # checkupdates, for waybar's custom/updates module
udiskie                 # removable media automount
libayatana-appindicator # udiskie's system tray icon
network-manager-applet  # nm-applet, network tray icon (autostart.lua)
blueman                 # blueman-applet, bluetooth tray icon (autostart.lua)
polkit-gnome            # authentication agent; without it auth_admin fails silently
gcr                     # gcr-ssh-agent.socket -> $XDG_RUNTIME_DIR/gcr/ssh
xdg-user-dirs           # xdg-user-dir, used by zsh/conf.d/env.zsh

# ── Applications ────────────────────────────────────────────────────────────
firefox                 # SUPER+B
pavucontrol             # waybar pulseaudio on-click
nm-connection-editor    # waybar network on-click

# ── Theming ─────────────────────────────────────────────────────────────────
ttf-jetbrains-mono-nerd # the font every component asks for by name
papirus-icon-theme      # row icons in the rofi window switcher
adwaita-icon-theme      # GTK_THEME=Adwaita:dark, XCURSOR_THEME=Adwaita
qt5ct                   # QT_QPA_PLATFORMTHEME=qt5ct

# ── Shell ───────────────────────────────────────────────────────────────────
zsh                     # the shell this config is for
git                     # required by zsh/plugin.zsh to bootstrap zinit
starship                # preferred prompt (conf.d/prompt.zsh)
fzf                     # zsh/completions/fzf.zsh, fzf-tab plugin
bat                     # zsh/functions/bat.zsh
eza                     # zsh/functions/eza.zsh
duf                     # zsh/functions/duf.zsh
neovim                  # $EDITOR / $VISUAL (zsh/.zshrc)
